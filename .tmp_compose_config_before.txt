name: co-working-ghost-system
services:
  postgres-db:
    container_name: coworking-postgres
    environment:
      POSTGRES_DB: coworking_db
      POSTGRES_PASSWORD: postgres
      POSTGRES_USER: postgres
    healthcheck:
      test:
        - CMD-SHELL
        - pg_isready -U postgres -d coworking_db
      timeout: 5s
      interval: 2s
      retries: 10
      start_period: 10s
    image: postgres:15
    networks:
      default: null
    ports:
      - mode: ingress
        target: 5432
        published: "5432"
        protocol: tcp
    restart: always
    volumes:
      - type: volume
        source: postgres_data
        target: /var/lib/postgresql/data
        volume: {}
  reservations-service:
    build:
      context: /tmp/workspace/antorlok/Co-Working-Ghost-System/services/reservations-service
      dockerfile: Dockerfile
    container_name: coworking-reservations-service
    depends_on:
      postgres-db:
        condition: service_healthy
        required: true
    environment:
      DATABASE_URL: postgres://postgres:postgres@postgres-db:5432/coworking_db
      PORT: "8003"
      RUST_LOG: reservations_service=debug,tower_http=debug,sqlx=warn
      SECRET_KEY: coworking_secret_key_change_me_in_prod
    networks:
      default: null
    ports:
      - mode: ingress
        target: 8003
        published: "8003"
        protocol: tcp
    restart: always
  users-service:
    build:
      context: /tmp/workspace/antorlok/Co-Working-Ghost-System/services/users-service
      dockerfile: Dockerfile
    container_name: coworking-users-service
    depends_on:
      postgres-db:
        condition: service_healthy
        required: true
    environment:
      DATABASE_URL: postgresql+psycopg2://postgres:postgres@postgres-db:5432/coworking_db
      SECRET_KEY: coworking_secret_key_change_me_in_prod
    networks:
      default: null
    ports:
      - mode: ingress
        target: 8000
        published: "8000"
        protocol: tcp
    restart: always

  default:
    name: co-working-ghost-system_default

  postgres_data:
    name: co-working-ghost-system_postgres_data
